# ▼ 必須 ------------------------------------------------------

# 作品名。一覧のカードと作品ページの見出しに表示される
title: "Meiousei Artist photo"

# 撮影日（YYYY-MM-DD）。一覧は新しい順に並び、年だけが表示される
# 日付が曖昧なら月初（-01）や年初（-01-01）でOK
date: 2025-03-01


# ▼ 任意（不要な項目は行ごと削除してよい） ---------------------

# 作品ページの冒頭に出る短い説明。1〜2文程度
# SNSでシェアされたときの説明文にも使われる
summary: "京都のバンド Meiousei のアーティスト写真。"

# 関連リンク（複数可）。label が表示名、url がリンク先
links:
  - label: "Meiousei 公式サイト"
    url: "https://example.com"
  - label: "Instagram"
    url: "https://www.instagram.com/example/"

# 分類。今は表示に使っていないが、将来の絞り込み機能用に付けておくと便利
# 例：live / artist / portrait / wedding / family / film / commercial
tags: [music, artist]

# 一覧のサムネイルにする写真のファイル名。省略すると 01.jpg（1枚目）
cover: 03.jpg

# 写真の並べ方
#   stack … 1枚ずつ縦に大きく並べる（省略時はこちら）
#   grid  … 2列で並べる。縦横混在でも隙間なく詰める（石垣状）。スナップ向き
#   rows  … 段組み。1段に perRow 枚ずつ、高さを揃えて横幅いっぱいに並べる
#           端数の段も横幅いっぱいになる（例: 8枚・perRow 3 → 3枚+3枚+2枚）
layout: stack

# rows のときの1段の枚数（省略時は 3）
perRow: 3

# rows の段ごとの枚数を個別に決めたいとき（省略可。指定すると perRow より優先）
# 例: 1段目3枚、2段目2枚、3段目1枚（大きく）、残りは perRow 枚ずつ
# rows: [3, 2, 1]

# true にすると、この作品はサイトに出ない（下書き・保留用）
draft: false
